html-comments

This host serves the canonical copy of the reviewer comment/suggestion overlay used by static HTML reports — html-comments.js and html-comments.css. Pages load them from here, so every document runs the same version and a fix reaches all of them at once.

<script>
window.HC_CONFIG = { endpoint: 'YOUR_APPS_SCRIPT_URL', project: 'UNIQUE-PROJECT-SLUG' };
var BASE = location.origin + '/';   // this host
var l = document.createElement('link');
l.rel = 'stylesheet'; l.href = BASE + 'html-comments.css';
document.head.appendChild(l);
var s = document.createElement('script');
s.src = BASE + 'html-comments.js';
document.head.appendChild(s);
</script>
